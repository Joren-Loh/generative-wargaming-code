   double RED_FIRE_RANGE  = __RED_FR__;
end_script_variables

platform_type FIGHTER WSF_PLATFORM
   icon f15c
   radar_signature FIGHTER_RCS

   mover WSF_AIR_MOVER
      maximum_speed        450 m/s
      minimum_speed         90 m/s
      maximum_altitude   16000 m
      maximum_radial_acceleration 9 g
      maximum_linear_acceleration 0.5 g
      update_interval      0.25 s
   end_mover

   weapon aam WSF_EXPLICIT_WEAPON
      launched_platform_type AAM
      weapon_effects AAM_EFFECT
      quantity        2
      firing_interval 2.0 sec
      salvo_interval  4.0 sec
      slew_mode       azimuth_and_elevation
      azimuth_slew_limits -180 deg 180 deg
   end_weapon

   processor shooter WSF_SCRIPT_PROCESSOR
      update_interval 0.5 s
      script_variables
         double min_range   = 2500.0;
         double fire_gap    = 6.0;
         double last_fire_t = -999.0;
      end_script_variables

      on_update
         if (TIME_NOW - last_fire_t < fire_gap) return;
         WsfWeapon w = PLATFORM.Weapon("aam");
         if (!w.IsValid()) return;
         if (w.QuantityRemaining() <= 0) return;

         double fr = RED_FIRE_RANGE;
         if (PLATFORM.Side() == "blue") fr = BLUE_FIRE_RANGE;

         string tgt_name = "";
         double best_r = 1.0e9;
         for (int i = 0; i < WsfSimulation.PlatformCount(); i += 1)
         {
            WsfPlatform p = WsfSimulation.PlatformEntry(i);
            if (!p.IsValid()) continue;
            if (p.Side() == PLATFORM.Side()) continue;
            if (p.Type() != "FIGHTER") continue;
            double r = PLATFORM.SlantRangeTo(p);
            if (r < best_r) { best_r = r; tgt_name = p.Name(); }
         }
         if (tgt_name == "") return;
         if (best_r < min_range || best_r > fr) return;

         WsfTrack track = WsfTrack();
         track.Initialize();
         track.SetTarget(tgt_name);
         if (w.Fire(track)) { last_fire_t = TIME_NOW; }
      end_on_update
   end_processor
end_platform_type
