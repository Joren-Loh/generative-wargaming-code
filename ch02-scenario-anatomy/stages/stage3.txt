# Section 2.2, step 3: add a Red platform and fly them at each other
log_file output/stage3.log

event_output
   file output/stage3.evt
   enable PLATFORM_ADDED
   enable PLATFORM_DELETED
end_event_output

platform_type FIGHTER WSF_PLATFORM
   mover WSF_AIR_MOVER
      maximum_speed                600 m/sec
      minimum_speed                120 m/sec
      default_radial_acceleration  6.0 g
   end_mover
end_platform_type

platform blue_1 FIGHTER
   side blue
   route
      position 30:00:00n 122:00:00e altitude 8000 m msl speed 250 m/sec
      position 30:00:00n 123:00:00e altitude 8000 m msl speed 250 m/sec
   end_route
end_platform

platform red_1 FIGHTER
   side red
   route
      position 30:00:00n 123:00:00e altitude 8000 m msl speed 250 m/sec
      position 30:00:00n 122:00:00e altitude 8000 m msl speed 250 m/sec
   end_route
end_platform

end_time 1 min
