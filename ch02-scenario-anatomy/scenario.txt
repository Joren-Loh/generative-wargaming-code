# ---------------- fighter platform type ----------------
platform_type FIGHTER WSF_PLATFORM
   icon F-16
   radar_signature FIGHTER_RCS

   mover WSF_AIR_MOVER
      maximum_speed                600 m/sec
      minimum_speed                120 m/sec
      default_radial_acceleration  6.0 g
      default_linear_acceleration  2.0 g
      at_end_of_path               extrapolate
   end_mover

   sensor radar WSF_GEOMETRIC_SENSOR
      on
      frame_time                   1 sec
      maximum_range                80 km
      azimuth_field_of_view        -60 deg 60 deg
      elevation_field_of_view      -30 deg 30 deg
      reports_location
      reports_velocity
      reports_range
      reports_bearing
      reports_iff
      ignore_same_side
      processor track_manager
   end_sensor

   processor track_manager WSF_TRACK_PROCESSOR
      on
      update_interval 1 sec
      raw_track_reporting on
   end_processor

   weapon aam AAM_LAUNCHER
      on
   end_weapon

   processor shooter WSF_SCRIPT_PROCESSOR
      on
      update_interval 1 sec
      on_update
         if (PLATFORM.Weapon("aam").QuantityRemaining() <= 0) { return; }
         foreach (WsfTrack track in PLATFORM.MasterTrackList())
         {
            double range = PLATFORM.SlantRangeTo(track);
            if (range < 20000.0 && PLATFORM.Weapon("aam").Fire(track))
            {
               writeln(TIME_NOW, " FIRE ", PLATFORM.Name(), " -> ", track.TargetName(), " R=", range);
               return;
            }
         }
      end_on_update
   end_processor
end_platform_type
