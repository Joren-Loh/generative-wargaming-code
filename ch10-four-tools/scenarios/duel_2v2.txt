# Run: mission.exe duel_2v2.txt, with cwd set to scenarios/
end_time 300 sec
random_seed 20261026

include_once short_range_ir_missile.txt

platform_type DUEL_FIGHTER WSF_PLATFORM
   icon f-18
   mover WSF_AIR_MOVER
   end_mover
   sensor eye WSF_GEOMETRIC_SENSOR
      on
      processor track_mgr
      azimuth_field_of_view -180 deg 180 deg
      elevation_field_of_view -90 deg 90 deg
      minimum_range 0 m
      maximum_range 70 km
      frame_time 1 sec
      ignore_same_side
      reports_location
      reports_velocity
      hits_to_establish_track 1 1
      hits_to_maintain_track 1 1
      track_quality 1.0
   end_sensor
   processor track_mgr WSF_TRACK_PROCESSOR
      purge_interval 60 sec
   end_processor
   processor shooter WSF_SCRIPT_PROCESSOR
      update_interval 1 sec
      script_variables
         double FIRE_RANGE = 12000.0;
         bool mFired = false;
      end_script_variables
      on_update
         if (mFired) { return; }
         WsfLocalTrackList tracks = PLATFORM.MasterTrackList();
         if (tracks.Size() == 0) { return; }
         WsfLocalTrack t = tracks.Entry(0);
         if (!t.IsValid()) { return; }
         if (PLATFORM.SlantRangeTo(t) > FIRE_RANGE) { return; }
         WsfWeapon w = PLATFORM.Weapon("srm");
         if (!w.IsValid()) { return; }
         w.CueToTarget(t);
         if (w.Fire(t)) { mFired = true; }
      end_on_update
   end_processor
   add weapon srm SHORT_RANGE_IR_MISSILE
      quantity 2
   end_weapon
end_platform_type
